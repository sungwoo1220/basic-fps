cmake_minimum_required(VERSION 3.21)
project(BasicFPS_DX11 LANGUAGES CXX)

enable_testing()
add_executable(ProtocolTests tests/ProtocolTests.cpp)
target_include_directories(ProtocolTests PRIVATE . ../unreal/Source/BasicFPS/Public)
target_compile_features(ProtocolTests PRIVATE cxx_std_17)
add_test(NAME dx11_protocol COMMAND ProtocolTests)

if(WIN32)
  enable_language(RC)
  add_executable(BasicFPS-DX11 WIN32 Trial.cpp Renderer.cpp Network.cpp Room.cpp Trial.rc)
  target_include_directories(BasicFPS-DX11 PRIVATE ../unreal/Source/BasicFPS/Public)
  target_compile_features(BasicFPS-DX11 PRIVATE cxx_std_17)
  target_compile_definitions(BasicFPS-DX11 PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(BasicFPS-DX11 PRIVATE d3d11 dxgi d3dcompiler d2d1 dwrite winhttp ws2_32 shell32 user32)
  if(MSVC)
    foreach(target BasicFPS-DX11 ProtocolTests)
      target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/utf-8;/W4;/permissive->")
      set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endforeach()
  endif()
endif()
